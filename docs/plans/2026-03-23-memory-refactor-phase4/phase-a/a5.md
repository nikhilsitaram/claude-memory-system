# A5: Implement RRF hybrid search in embeddings.py

**Avoid:** Do not use learned fusion weights or cross-encoder reranking -- RRF with k=60 is the design choice (simple, proven, no training needed). Do not remove the existing `search_similar` function -- `search_hybrid` wraps it as one input source. Do not import from storage for FTS5 at module level -- use a lazy import inside `search_hybrid` since FTS5 helpers might not exist in older storage.py versions. The function must degrade gracefully: if vector is unavailable, use FTS5 only; if FTS5 is unavailable, use vector only; if neither, fall back to SQL ranked.

### Step 2: Implement search_hybrid in embeddings.py

Add constants and the function after the existing `search_similar`:

```python
RRF_K = 60  # RRF constant (standard value from Cormack et al.)


def search_hybrid(
    conn,
    query: str,
    scope: Optional[str] = None,
    top_k: int = DEFAULT_TOP_K,
) -> list[ScoredDataPoint]:
    """Hybrid search combining FTS5 BM25 and vector KNN via Reciprocal Rank Fusion.

    Fallback chain:
    - Both FTS5 + vector available -> RRF hybrid
    - Vector only -> vector search
    - FTS5 only -> BM25 keyword search
    - Neither -> SQL ranked fallback (salience + recency)

    Returns sorted list of ScoredDataPoint (highest score first), limited to top_k.
    """
    from storage import fts_search, _row_to_data_point, _DP_COLUMNS

    has_fts = _has_table(conn, "fts_data")
    has_vec = HAS_FASTEMBED and HAS_SQLITE_VEC and _has_table(conn, "vec_data")

    fts_results = []
    vec_results = []

    # FTS5 path
    if has_fts:
        try:
            fts_hits = fts_search(conn, query, scope=scope, limit=top_k * 2)
            fts_results = [(hit["data_point_id"], rank_idx) for rank_idx, hit in enumerate(fts_hits)]
        except Exception:
            pass

    # Vector path
    if has_vec:
        try:
            vec_results_raw = search_similar(conn, query, top_k=top_k * 2, scope=scope)
            vec_results = [(r.data_point.id, rank_idx) for rank_idx, r in enumerate(vec_results_raw)]
        except Exception:
            pass

    if fts_results and vec_results:
        # RRF fusion
        scores = {}
        for dp_id, rank in fts_results:
            scores[dp_id] = scores.get(dp_id, 0.0) + 1.0 / (RRF_K + rank + 1)
        for dp_id, rank in vec_results:
            scores[dp_id] = scores.get(dp_id, 0.0) + 1.0 / (RRF_K + rank + 1)

        ranked_ids = sorted(scores.keys(), key=lambda x: scores[x], reverse=True)[:top_k]
        return _fetch_scored_data_points(conn, ranked_ids, scores)

    elif vec_results:
        ranked_ids = [dp_id for dp_id, _ in vec_results[:top_k]]
        scores = {dp_id: 1.0 / (RRF_K + rank + 1) for dp_id, rank in vec_results}
        return _fetch_scored_data_points(conn, ranked_ids, scores)

    elif fts_results:
        ranked_ids = [dp_id for dp_id, _ in fts_results[:top_k]]
        scores = {dp_id: 1.0 / (RRF_K + rank + 1) for dp_id, rank in fts_results}
        return _fetch_scored_data_points(conn, ranked_ids, scores)

    else:
        # SQL ranked fallback
        return _sql_ranked_fallback(conn, scope, top_k)


def _fetch_scored_data_points(conn, ranked_ids, scores):
    """Fetch data_points by ID and return as ScoredDataPoint list.

    Applies salience + recency modulation using the existing score_memory()
    function. Note: score_memory(vec_distance, chunk) takes a vec_distance
    and a chunk/data_point object. For RRF-fused results where no single
    vec_distance exists, pass 0.5 as a neutral distance.
    """
    from storage import _row_to_data_point, _DP_COLUMNS
    results = []
    for dp_id in ranked_ids:
        row = conn.execute(
            f"SELECT {_DP_COLUMNS} FROM data_points WHERE id = ?", (dp_id,)
        ).fetchone()
        if row:
            dp = _row_to_data_point(row)
            # score_memory expects (vec_distance, chunk_or_dp) -- use 0.5 as
            # neutral distance since RRF already incorporates vector ranking
            salience_recency = score_memory(0.5, dp)
            final_score = scores[dp_id] + salience_recency
            results.append(ScoredDataPoint(data_point=dp, score=final_score, vec_similarity=0.0))
    results.sort(key=lambda x: x.score, reverse=True)
    return results


def _sql_ranked_fallback(conn, scope, top_k):
    """Fallback search using salience + recency ordering (no semantic understanding)."""
    from storage import _row_to_data_point, _DP_COLUMNS
    query = f"SELECT {_DP_COLUMNS} FROM data_points WHERE type = 'memory' AND salience > 0.05"
    params = []
    if scope:
        query += " AND scope = ?"
        params.append(scope)
    query += " ORDER BY salience DESC, created_at DESC LIMIT ?"
    params.append(top_k)

    results = []
    for row in conn.execute(query, params).fetchall():
        dp = _row_to_data_point(row)
        results.append(ScoredDataPoint(data_point=dp, score=dp.salience, vec_similarity=0.0))
    return results
```
