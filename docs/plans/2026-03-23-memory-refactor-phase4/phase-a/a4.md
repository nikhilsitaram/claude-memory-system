# A4: Add FTS5 virtual table with migration backfill

**Avoid:** Do not use external search libraries (Whoosh, Tantivy) -- FTS5 is built into SQLite and requires zero dependencies. Do not create FTS5 as a content table synced via triggers -- use an external content table approach where we manually sync inserts/deletes. This gives us explicit control over when the index updates. Do not use `tokenize='unicode61'` alone -- add `'porter unicode61'` for English stemming (so "running" matches "run"). The FTS5 table must live in the same `memory.db` file (not a separate database).

### Step 2: Add FTS5 DDL and helper functions to storage.py

Add the FTS5 DDL constant:

```python
FTS5_DDL = """\
CREATE VIRTUAL TABLE IF NOT EXISTS fts_data USING fts5(
    content,
    data_point_id UNINDEXED,
    scope UNINDEXED,
    tokenize='porter unicode61'
);
"""
```

Add the `_ensure_fts_table` function (called from `ensure_db`):

```python
def _ensure_fts_table(conn) -> None:
    """Create FTS5 table and backfill from data_points if needed."""
    conn.executescript(FTS5_DDL)

    # Backfill: insert data_points not already in fts_data
    conn.execute("""
        INSERT INTO fts_data(content, data_point_id, scope)
        SELECT dp.content, dp.id, dp.scope
        FROM data_points dp
        WHERE dp.content IS NOT NULL
        AND dp.id NOT IN (SELECT data_point_id FROM fts_data)
    """)
    conn.commit()
```

Add FTS5 CRUD helpers:

```python
def fts_insert(conn, dp_id: str, content: str, scope: str | None) -> None:
    """Insert a data_point's content into the FTS5 index."""
    if not content:
        return
    conn.execute(
        "INSERT INTO fts_data(content, data_point_id, scope) VALUES (?, ?, ?)",
        (content, dp_id, scope)
    )


def fts_delete(conn, dp_id: str) -> None:
    """Remove a data_point from the FTS5 index."""
    conn.execute("DELETE FROM fts_data WHERE data_point_id = ?", (dp_id,))


def fts_search(conn, query: str, scope: str | None, limit: int = 10) -> list[dict]:
    """Search FTS5 index with BM25 ranking.

    Returns list of dicts with 'data_point_id', 'scope', 'rank' keys,
    ordered by BM25 relevance (best first).
    """
    if not query or not query.strip():
        return []

    # Sanitize query for FTS5: escape special chars, use implicit AND
    safe_query = query.replace('"', '""')
    terms = safe_query.split()
    fts_query = " ".join(f'"{t}"' for t in terms if t.strip())
    if not fts_query:
        return []

    if scope:
        rows = conn.execute(
            "SELECT data_point_id, scope, rank FROM fts_data "
            "WHERE fts_data MATCH ? AND scope = ? "
            "ORDER BY rank LIMIT ?",
            (fts_query, scope, limit)
        ).fetchall()
    else:
        rows = conn.execute(
            "SELECT data_point_id, scope, rank FROM fts_data "
            "WHERE fts_data MATCH ? ORDER BY rank LIMIT ?",
            (fts_query, limit)
        ).fetchall()

    return [{"data_point_id": r[0], "scope": r[1], "rank": r[2]} for r in rows]
```
