# C2: Integrate consolidation into synthesis_cron.py with daily gate

**Avoid:** Do not create a separate launchd/systemd timer for consolidation -- it piggybacks on the existing synthesis_cron scheduling. Do not run consolidation before synthesis completes -- consolidation is a post-step. Do not use a file-based timestamp for last_consolidation -- use a metadata key in the DB (the DB is the source of truth). Do not consolidate if synthesis failed -- only run after a successful synthesis pass.

### Step 2: Add consolidation gate and integration to synthesis_cron.py

Add the gate functions:

```python
def _should_consolidate(conn, settings) -> bool:
    """Check if consolidation should run based on interval and memory count."""
    consol = settings.get("consolidation", {})
    interval_hours = consol.get("intervalHours", 24)
    min_memories = consol.get("minMemories", 5)

    # Check last_consolidation timestamp
    row = conn.execute(
        "SELECT value FROM metadata WHERE key = 'last_consolidation'"
    ).fetchone()

    if row and row[0]:
        from datetime import datetime, timezone
        try:
            last = datetime.fromisoformat(row[0].replace("Z", "+00:00"))
            elapsed = (datetime.now(timezone.utc) - last).total_seconds() / 3600
            if elapsed < interval_hours:
                return False
        except (ValueError, AttributeError):
            pass

    # Check active unconsolidated memory count
    count_row = conn.execute(
        "SELECT COUNT(*) FROM data_points WHERE type = 'memory' AND salience > 0.1 "
        "AND source_type != 'consolidation'"
    ).fetchone()
    if not count_row or count_row[0] < min_memories:
        return False

    return True


def _is_backfill(conn) -> bool:
    """Check if this is the first-ever consolidation run."""
    row = conn.execute(
        "SELECT value FROM metadata WHERE key = 'last_consolidation'"
    ).fetchone()
    return row is None or row[0] is None


def _update_consolidation_timestamp(conn):
    """Update the last_consolidation metadata key."""
    now = datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")
    conn.execute(
        "INSERT OR REPLACE INTO metadata (key, value) VALUES ('last_consolidation', ?)",
        (now,)
    )
    conn.commit()
```

Note: This requires a `metadata` table. Add to storage.py:
```sql
CREATE TABLE IF NOT EXISTS metadata (
    key TEXT PRIMARY KEY,
    value TEXT
);
```

Add the consolidation post-step to the main synthesis flow, called after `_run_synthesis_v3`:

```python
def _run_consolidation_post_step(conn, settings):
    """Run consolidation as a post-step after synthesis, if gate conditions met."""
    if not _should_consolidate(conn, settings):
        return

    try:

        backfill = _is_backfill(conn)
        stats = run_consolidation(conn, settings=settings, backfill=backfill)
```
