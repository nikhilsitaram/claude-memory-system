# A4: Scoring function and search_similar

**Avoid:** Do not use `datetime.utcnow()` -- it is deprecated in Python 3.12+. Use `datetime.now(timezone.utc)` for all current-time calculations. Do not divide by zero in `days_since()` -- if `last_accessed` and `created_at` are both None, use current time as fallback (0 days since = max recency). Do not assume `chunk.salience` is always set -- default to `1.0` when None (Phase 1 fallback per design doc). Do not assume `chunk.last_accessed` is set -- fall back to `chunk.created_at`, then to current time.

**Avoid:** Do not fetch all vectors from vec_chunks -- use sqlite-vec's built-in KNN query (`SELECT ... FROM vec_chunks WHERE embedding MATCH ? ORDER BY distance LIMIT ?`). The `MATCH` syntax is specific to sqlite-vec virtual tables. The distance returned is L2 distance by default; for cosine similarity, the design doc uses `1 - distance` where distance is already a cosine distance (sqlite-vec with `vec0` uses cosine distance when vectors are normalized -- FastEmbed's all-MiniLM-L6-v2 produces normalized vectors).

```python
def _days_since(iso_date: str | None, fallback: str | None = None) -> float:
    """Calculate days elapsed since an ISO date string.

    Args:
        iso_date: ISO 8601 date string (e.g., '2026-03-20T12:00:00+00:00').
        fallback: Fallback ISO date if iso_date is None.

    Returns:
        Number of days since the date. Returns 0.0 if both are None.
    """
    date_str = iso_date or fallback
    if not date_str:
        return 0.0
    try:
        dt = datetime.fromisoformat(date_str)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        now = datetime.now(timezone.utc)
        delta = now - dt
        return max(0.0, delta.total_seconds() / 86400.0)
    except (ValueError, TypeError):
        return 0.0


def score_memory(vec_distance: float, chunk: ChunkRow) -> float:
    """Compute composite relevance score for a memory chunk.

    Combines vector similarity (saturating exponential boost),
    recency decay, and salience into a weighted score.

    Phase 1 fallbacks:
    - last_accessed defaults to created_at (or current time if both None)
    - salience defaults to 1.0

    Formula:
        vec_sim = 1.0 - vec_distance
        boosted = 1 - exp(-VEC_BOOST_RATE * vec_sim)
        recency = exp(-RECENCY_DECAY * days_since(last_accessed))
        score = VEC_SIM_WEIGHT * boosted + RECENCY_WEIGHT * recency + SALIENCE_WEIGHT * salience

    Args:
        vec_distance: Distance from sqlite-vec (0.0 = identical, 1.0+ = distant).
        chunk: ChunkRow with metadata for recency and salience.

    Returns:
        Composite score in [0, 1] range.
    """
    vec_sim = max(0.0, 1.0 - vec_distance)
    boosted = 1.0 - math.exp(-VEC_BOOST_RATE * vec_sim)

    days = _days_since(
        getattr(chunk, "last_accessed", None),
        getattr(chunk, "created_at", None),
    )
    recency = math.exp(-RECENCY_DECAY * days)

    salience = getattr(chunk, "salience", None)
    if salience is None:
        salience = 1.0

    return VEC_SIM_WEIGHT * boosted + RECENCY_WEIGHT * recency + SALIENCE_WEIGHT * salience
```

### Step 2: Replace the `search_similar` stub

```python
def search_similar(
    conn: sqlite3.Connection,
    query: str,
    top_k: int = DEFAULT_TOP_K,
    scope: str | None = None,
) -> list[ScoredChunk]:
    """Semantic search for memory chunks similar to a query.

    Embeds the query, searches vec_chunks via KNN, JOINs to chunks
    for metadata, optionally filters by scope, scores with composite
    formula, and returns top_k results ranked by score.

    Args:
        conn: Open SQLite connection with vec_chunks and chunks tables.
        query: Natural language query to search for.
        top_k: Maximum number of results to return.
        scope: Optional scope filter (e.g., 'global', 'project-name').

    Returns:
        List of ScoredChunk, sorted by score descending.
        Empty list if fastembed/sqlite-vec unavailable or no results.
    """
    if not HAS_FASTEMBED or not HAS_SQLITE_VEC:
        return []

    query_vec = embed_text(query)
    if not query_vec:
        return []

    fetch_k = top_k * 3

    # Prefix chunk columns with table alias to avoid ambiguity
    # (vec_chunks also has a source_type auxiliary column).
    qualified_columns = ", ".join(f"c.{col.strip()}" for col in _CHUNK_COLUMNS.split(","))

    try:
        rows = conn.execute(
            f"SELECT vc.distance, {qualified_columns} "
            f"FROM vec_chunks vc "
            f"JOIN chunks c ON vc.chunk_id = c.id "
            f"WHERE vc.embedding MATCH ? "
            f"ORDER BY vc.distance "
            f"LIMIT ?",
            (_serialize_vector(query_vec), fetch_k),
        ).fetchall()
    except Exception:
        return []

    if not rows:
        return []

    scored = []
    for row in rows:
        distance = row[0]
        chunk = _row_to_chunk(row[1:])

        if scope and chunk.scope != scope:
            continue

        score = score_memory(distance, chunk)
        scored.append(ScoredChunk(
            chunk=chunk,
            score=score,
            vec_similarity=max(0.0, 1.0 - distance),
        ))

    scored.sort(key=lambda s: s.score, reverse=True)
    return scored[:top_k]
```
