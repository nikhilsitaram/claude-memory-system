```python
def apply_memory_ops(
    ops: list[dict],
    session_date: str,
    ltm_dir: Path | None = None,
    global_file: Path | None = None,
) -> list[str]:
    """Apply CRUD operations from ===MEMORY_OPS=== to DB and markdown.

    Args:
        ops: List of operation dicts from parsed MEMORY_OPS.
        session_date: Date string (YYYY-MM-DD) for new entries.
        ltm_dir: Project LTM directory. Defaults to get_project_memory_dir().
        global_file: Global LTM file. Defaults to get_global_memory_file().

    Returns:
        List of warning strings (empty if all ops succeeded).
    """
    if not ops:
        return []

    from storage import (
        ensure_db, close_db, insert_chunk, insert_node, insert_edge,
        update_chunk_content, query_chunk_by_id, update_chunk_salience,
        invalidate_edge, ChunkRow, NodeRow,
    )

    if global_file is None:
        global_file = get_global_memory_file()
    if ltm_dir is None:
        ltm_dir = get_project_memory_dir()

    warnings = []
    conn = None
    try:
        conn = ensure_db()
        for op in ops:
            action = op.get("action", "").upper()
            if action == "ADD":
                warnings.extend(_apply_add(conn, op, session_date, ltm_dir, global_file))
            elif action == "UPDATE":
                warnings.extend(_apply_update(conn, op, ltm_dir, global_file))
            elif action == "DELETE":
                warnings.extend(_apply_delete(conn, op, session_date, ltm_dir, global_file))
            elif action == "NOOP":
                _apply_noop(conn, op)
            else:
                warnings.append(f"Unknown MEMORY_OPS action: {action}")
        conn.commit()
    except Exception as e:
        warnings.append(f"MEMORY_OPS apply error: {e}")
        if conn:
            conn.rollback()
    finally:
        if conn:
            close_db(conn)

    return warnings
```

```python
def _apply_add(conn, op, session_date, ltm_dir, global_file) -> list[str]:
    """Handle ADD: insert chunk+node in DB, append to LTM markdown."""
    warnings = []
    fact = op.get("fact", "")
    scope = op.get("scope", "global")
    section = op.get("section", "Key Actions")
    entities = op.get("entities")
    entry_type = op.get("type", "implement")

    # Insert chunk into DB
    chunk = ChunkRow(
        content=fact,
        source_file=f"{scope}-long-term-memory.md",
        source_type="ltm",
        scope=scope,
        section=f"## {section}",
        entry_type=entry_type,
        created_at=session_date,
        entities=json.dumps(entities) if entities else None,
    )
    chunk_id = insert_chunk(conn, chunk)

    # Append to LTM markdown
    entry_line = f"- ({session_date}) [{entry_type}] {fact}"
    target_file = global_file if scope == "global" else ltm_dir / project_name_to_filename(scope)
    if target_file.exists():
        _append_entry_to_section(target_file, f"## {section}", entry_line)
    else:
        warnings.append(f"ADD: LTM file not found for scope '{scope}', DB-only")

    return warnings


def _apply_update(conn, op, ltm_dir, global_file) -> list[str]:
    """Handle UPDATE: modify chunk content+entities in DB, update markdown line."""
    warnings = []
    chunk_id = op.get("id")
    new_fact = op.get("fact", "")
    new_entities = op.get("entities")

    if not chunk_id:
        warnings.append("UPDATE: missing chunk id")
        return warnings

    existing = query_chunk_by_id(conn, chunk_id)
    if not existing:
        warnings.append(f"UPDATE: chunk {chunk_id} not found in DB")
        return warnings

    # Update DB
    entities_json = json.dumps(new_entities) if new_entities else None
    update_chunk_content(conn, chunk_id, new_fact, entities_json)

    # Update markdown (match by content_hash, fallback to substring)
    target_file = global_file if existing.scope == "global" else ltm_dir / project_name_to_filename(existing.scope)
    if target_file.exists():
        updated = _update_markdown_line(target_file, existing.content_hash, existing.content, new_fact)
        if not updated:
            warnings.append(f"UPDATE: markdown line not found for chunk {chunk_id}, DB-only")

    return warnings

def _apply_noop(conn, op) -> None:
    """Handle NOOP: increment evidence_count."""
    chunk_id = op.get("id")
    if chunk_id:
        conn.execute(
            "UPDATE chunks SET evidence_count = evidence_count + 1 WHERE id = ?",
            (chunk_id,),
        )
```
