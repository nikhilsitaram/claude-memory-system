# C5: Add bi-temporal edge handling for contradiction detection

**Avoid:** Do not delete rows from the edges table -- the entire point of bi-temporal tracking is to preserve history. Set `valid_to` and `expired_at` instead. Do not create edges for ADD operations in this task -- edge creation for new facts is a Phase 3/4 concern (triplet embeddings). This task focuses on invalidating existing edges when DELETE operations detect contradictions, and creating fresh replacement edges when the DELETE supplies a new fact. Do not change the edges table schema -- `valid_from`, `valid_to`, `expired_at` columns already exist.

### Step 2: Add temporal query helpers to storage.py

Add to `scripts/storage.py`:

```python
_EDGE_COLUMNS = (
    "id, source, target, type, fact, properties, created_at, "
    "valid_from, valid_to, expired_at, weight, source_sessions"
)


def _row_to_edge(row: tuple) -> EdgeRow:
    """Convert a raw SQLite row tuple to an EdgeRow dataclass."""
    return EdgeRow(
        id=row[0], source=row[1], target=row[2], type=row[3],
        fact=row[4], properties=row[5], created_at=row[6],
        valid_from=row[7], valid_to=row[8], expired_at=row[9],
        weight=row[10], source_sessions=row[11],
    )


def query_current_edges(conn: sqlite3.Connection) -> list[EdgeRow]:
    """Query all currently valid edges (valid_to IS NULL)."""
    rows = conn.execute(
        f"SELECT {_EDGE_COLUMNS} FROM edges WHERE valid_to IS NULL"
    ).fetchall()
    return [_row_to_edge(r) for r in rows]


def query_edges_at_date(
    conn: sqlite3.Connection, date_str: str
) -> list[EdgeRow]:
    """Query edges that were valid at a specific date.

    Returns edges where valid_from <= date AND (valid_to IS NULL OR valid_to > date).
    """
    rows = conn.execute(
        f"SELECT {_EDGE_COLUMNS} FROM edges "
        f"WHERE (valid_from IS NULL OR valid_from <= ?) "
        f"AND (valid_to IS NULL OR valid_to > ?)",
        (date_str, date_str),
    ).fetchall()
    return [_row_to_edge(r) for r in rows]


def query_edges_for_node(
    conn: sqlite3.Connection, node_id: str
) -> list[EdgeRow]:
    """Query all edges (valid and invalid) connected to a node."""
    rows = conn.execute(
        f"SELECT {_EDGE_COLUMNS} FROM edges "
        f"WHERE source = ? OR target = ?",
        (node_id, node_id),
    ).fetchall()
    return [_row_to_edge(r) for r in rows]
```

```python
def _apply_delete(conn, op, session_date, ltm_dir, global_file) -> list[str]:
    """Handle DELETE: bi-temporal edge invalidation, salience=0, archive markdown."""
    from storage import (
        query_chunk_by_id, update_chunk_salience, invalidate_edge,
        query_edges_for_node, query_node_by_name_and_type, insert_edge, EdgeRow,
    )

    warnings = []
    chunk_id = op.get("id")
    reason = op.get("reason", "")

    if not chunk_id:
        warnings.append("DELETE: missing chunk id")
        return warnings

    existing = query_chunk_by_id(conn, chunk_id)
    if not existing:
        warnings.append(f"DELETE: chunk {chunk_id} not found in DB")
        return warnings

    now_iso = datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")

    # Find and invalidate edges connected to this chunk's nodes.
    # Node linkage: match nodes whose name appears in the chunk content
    # (not all nodes in scope -- that would over-invalidate).
    # Extract entity names from the chunk's entities JSON if available,
    # otherwise fall back to keyword matching against node names.
    import json as _json
    chunk_entities = []
    if existing.entities:
        try:
            chunk_entities = _json.loads(existing.entities)
        except (ValueError, TypeError):
            pass

    # Find nodes related to this specific chunk via entity matching
    related_node_ids = set()
    if chunk_entities:
        for entity_name in chunk_entities:
            node = query_node_by_name_and_type(conn, entity_name, "entity")
            if node:
                related_node_ids.add(node.id)

    for node_id in related_node_ids:
        edges = query_edges_for_node(conn, node_id)
        for edge in edges:
            if edge.valid_to is None:  # Only invalidate current edges
                invalidate_edge(conn, edge.id, valid_to=session_date, expired_at=now_iso)

    # Set chunk salience to 0
    update_chunk_salience(conn, chunk_id, 0.0)

    # Archive in markdown
    target_file = (global_file if existing.scope == "global"
                   else ltm_dir / project_name_to_filename(existing.scope))
    if target_file.exists():
        _archive_markdown_line(target_file, existing.content_hash, existing.content)

    return warnings
```
