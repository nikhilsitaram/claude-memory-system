**Avoid:** Do not import storage.py at module level in load_memory.py -- the DB may not exist yet (first run before install.py). Use lazy imports inside the tracking function. Do not block SessionStart if the DB is unavailable or locked -- access tracking is best-effort; a failed write should log a warning to stderr and continue. Do not use `conn.execute("BEGIN")` -- use `conn.execute("BEGIN IMMEDIATE")` to prevent write conflicts with concurrent sessions.

### Step 2: Implement track_memory_access function

Add to `scripts/load_memory.py`:

```python
# Constants for access tracking
REINFORCEMENT_ETA = 0.18  # OpenMemory's tuned reinforcement rate
MAX_BUSY_RETRIES = 3
BUSY_RETRY_DELAY = 0.1  # seconds


def track_memory_access(chunk_ids: list[str], node_ids: list[str] | None = None) -> None:
    """Record access for served chunks/nodes in the DB.

    Best-effort: failures are logged to stderr but do not block SessionStart.
    Uses BEGIN IMMEDIATE to prevent write conflicts with concurrent sessions.

    Args:
        chunk_ids: List of chunk IDs that were served this session.
        node_ids: Optional list of node IDs that were served.
    """
    if not chunk_ids and not node_ids:
        return

    try:
        from storage import (
            batch_update_access,
            get_db,
            close_db,
            update_chunk_salience,
            update_node_salience,
            query_chunks_with_salience,
        )
    except ImportError:
        return  # Storage module not available

    conn = None
    try:
        conn = get_db()
        _execute_with_retry(conn, chunk_ids, node_ids or [])
    except FileNotFoundError:
        pass  # DB doesn't exist yet
    except Exception as e:
        print(f"Warning: Access tracking failed: {e}", file=sys.stderr)
    finally:
        if conn:
            close_db(conn)


def _execute_with_retry(conn, chunk_ids, node_ids):
    """Execute access tracking with BEGIN IMMEDIATE and retry on SQLITE_BUSY."""
    import time
    from storage import (
        batch_update_access,
        update_chunk_salience,
        update_node_salience,
    )

    for attempt in range(MAX_BUSY_RETRIES):
        try:
            conn.execute("BEGIN IMMEDIATE")
            break
        except sqlite3.OperationalError as e:
            if "database is locked" in str(e) and attempt < MAX_BUSY_RETRIES - 1:
                time.sleep(BUSY_RETRY_DELAY)
                continue
            raise

    try:
        now = datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")

        # Increment access counts
        batch_update_access(conn, chunk_ids, timestamp=now)

        # Reinforce salience with diminishing returns
        for cid in chunk_ids:
            row = conn.execute(
                "SELECT salience FROM chunks WHERE id = ?", (cid,)
            ).fetchone()
            if row:
                current = row[0] or 1.0
                new_salience = min(1.0, current + REINFORCEMENT_ETA * (1.0 - current))
                update_chunk_salience(conn, cid, new_salience)

        # Same for nodes
        for nid in (node_ids or []):
            row = conn.execute(
                "SELECT salience FROM nodes WHERE id = ?", (nid,)
            ).fetchone()
            if row:
                current = row[0] or 1.0
                new_salience = min(1.0, current + REINFORCEMENT_ETA * (1.0 - current))
                update_node_salience(conn, nid, new_salience)

        conn.commit()
    except Exception:
        conn.rollback()
        raise
```

### Step 3: Wire access tracking into main()

In `load_memory.py`'s `main()` function, after all memory sections are loaded and printed, call `track_memory_access()` with the IDs of chunks/nodes that were served. This requires collecting chunk IDs during the load phase.

The integration point is after `print("</memory>")` -- add:

```python
# Best-effort access tracking (non-blocking)
if served_chunk_ids:
    track_memory_access(served_chunk_ids)
```

The `served_chunk_ids` list is populated by modifying `load_global_memory()`, `load_project_memory()`, etc. to also query the DB for matching chunk IDs. This is a lightweight query -- we already know the scope and source_file.
