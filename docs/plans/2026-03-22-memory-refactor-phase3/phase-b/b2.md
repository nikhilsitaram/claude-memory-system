### Step 2: Implement search_memories in memory_server.py

```python
async def _search_memories(query, scope=None, top_k=10):
    conn = _db_conn
    if _model_ready.is_set():
        # Vector search path
        results = search_similar(conn, query, top_k=top_k, scope=scope)
    else:
        # SQL fallback: rank by salience + recency
        results = _sql_ranked_search(conn, scope, top_k)

    # Apply graph boost
    result_ids = {r.data_point.id for r in results}
    for r in results:
        edges = query_edges_for_node(conn, r.data_point.id)
        connected = sum(1 for e in edges
                       if (e.source in result_ids or e.target in result_ids)
                       and e.valid_to is None)
        boost = min(0.15, connected * 0.05)
        r.score += boost

    # Re-sort after boost
    results.sort(key=lambda x: x.score, reverse=True)

    # Format response with provenance chains
    # Uses simple edge query for provenance (full recursive CTE in C4 later)
    formatted = []
    for r in results[:top_k]:
        prov_edges = conn.execute(
            "SELECT id, type, reason FROM edges WHERE source=? AND type IN "
            "('supersedes','contradicts','refines','led_to','supports') AND valid_to IS NULL",
            (r.data_point.id,)
        ).fetchall()
        formatted.append({
            "id": r.data_point.id,
            "content": r.data_point.content,
            "score": round(r.score, 3),
            "scope": r.data_point.scope,
            "entities": json.loads(r.data_point.entities) if r.data_point.entities else [],
            "provenance": [{"id": e[0], "type": e[1], "reason": e[2]} for e in prov_edges],
        })
    return formatted
```
