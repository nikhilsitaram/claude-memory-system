**Avoid:** Do not create entity data_points without checking if they already exist -- use upsert pattern (query by name+type, create only if missing). All DB operations must be in a single transaction so failure rolls back cleanly.

### Step 2: Implement _write_memory in memory_server.py

```python
async def _write_memory(fact, scope, salience=None, entities=None,
                        supersedes=None, relation_type=None, relation_reason=None):
    conn = _db_conn
    try:
        conn.execute("BEGIN IMMEDIATE")

        # Auto-pin user scope
        if scope == "user":
            salience = 1.0
            consolidated = 1
        else:
            salience = salience or 0.7
            consolidated = 0

        # Create memory data_point
        dp = DataPointRow(
            type="memory", content=fact, scope=scope,
            salience=salience, consolidated=consolidated,
            source_type="manual",
            created_at=datetime.now(timezone.utc).isoformat(),
        )
        dp_id = insert_data_point(conn, dp)

        # Generate and index embedding
        vec = embed_text(fact)
        if vec:
            conn.execute(
                "INSERT INTO vec_data (embedding, data_point_id, type) VALUES (?, ?, ?)",
                (_serialize_vector(vec), dp_id, "memory"),
            )

        # Create entity data_points + mentions edges
        if entities:
            for entity_name in entities:
                entity_id = _get_or_create_entity(conn, entity_name, scope)
                insert_edge(conn, EdgeRow(
                    source=dp_id, target=entity_id, type="mentions",
                    created_at=dp.created_at,
                ))

        # Provenance edge (uses insert_edge directly; C4 adds convenience wrapper later)
        if supersedes:
            edge_type = relation_type or "supersedes"
            insert_edge(conn, EdgeRow(
                source=dp_id, target=supersedes, type=edge_type,
                reason=relation_reason, created_at=dp.created_at,
                valid_from=dp.created_at,
            ))

        conn.commit()
        return {"id": dp_id, "status": "created"}
    except Exception:
        conn.rollback()
        raise
```
