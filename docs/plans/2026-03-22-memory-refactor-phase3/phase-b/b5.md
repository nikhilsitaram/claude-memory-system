# B5: Implement traverse_graph tool

**Avoid:** Do not use Python-level recursion for graph traversal -- use a SQL recursive CTE which is more efficient and naturally handles cycle detection. Do not return raw edge rows -- format as a tree/list with human-readable relationship descriptions. Do not traverse expired edges (where `valid_to IS NOT NULL`) -- only follow currently valid connections.

### Step 2: Implement _traverse_graph using recursive CTE

```python
async def _traverse_graph(entity, depth=2, relationship_type=None):
    conn = _db_conn

    # Build recursive CTE query
    type_filter = "AND e.type = ?" if relationship_type else ""
    params = [entity, entity]
    if relationship_type:
        params.extend([relationship_type])

    query = f"""
    WITH RECURSIVE graph_walk(dp_id, depth, path, edge_type, edge_reason) AS (
        -- Base case: direct neighbors
        SELECT
            CASE WHEN e.source = ? THEN e.target ELSE e.source END,
            1,
            e.id,
            e.type,
            e.reason
        FROM edges e
        WHERE (e.source = ? OR e.target = ?)
          AND e.valid_to IS NULL
          {type_filter}

        UNION ALL

        -- Recursive step
        SELECT
            CASE WHEN e.source = gw.dp_id THEN e.target ELSE e.source END,
            gw.depth + 1,
            gw.path || ',' || e.id,
            e.type,
            e.reason
        FROM graph_walk gw
        JOIN edges e ON (e.source = gw.dp_id OR e.target = gw.dp_id)
        WHERE gw.depth < ?
          AND e.valid_to IS NULL
          {type_filter}
    )
    SELECT DISTINCT gw.dp_id, gw.depth, gw.edge_type, gw.edge_reason,
           dp.content, dp.type, dp.name, dp.scope
    FROM graph_walk gw
    JOIN data_points dp ON dp.id = gw.dp_id
    ORDER BY gw.depth, dp.salience DESC
    """

    params_full = [entity, entity, entity]
    if relationship_type:
        params_full.append(relationship_type)
    params_full.append(depth)
    if relationship_type:
        params_full.append(relationship_type)

    rows = conn.execute(query, params_full).fetchall()
    # Format results
    return [{"id": r[0], "depth": r[1], "relationship": r[2],
             "reason": r[3], "content": r[4], "type": r[5],
             "name": r[6], "scope": r[7]} for r in rows]
```
