# B4: Implement delete_memory tool

**Avoid:** Do not hard-delete the data_point row -- set salience=0 (soft delete). Do not skip provenance -- even pure deletions (no replacement) need a deletion marker data_point with a `supersedes` edge to preserve the audit trail. Do not invalidate ALL edges for the target -- only invalidate edges where the deleted data_point is the source or target. Return the deleted content in the response so Claude can acknowledge what was removed.

### Step 2: Implement _delete_memory

```python
async def _delete_memory(id, reason=None):
    conn = _db_conn
    target = query_data_point_by_id(conn, id)
    if not target:
        return {"error": f"Data point {id} not found"}

    try:
        conn.execute("BEGIN IMMEDIATE")

        # Soft delete
        soft_delete_data_point(conn, id)

        # Invalidate connected edges
        now = datetime.now(timezone.utc).isoformat()
        edges = query_edges_for_node(conn, id)
        for edge in edges:
            if edge.valid_to is None:
                invalidate_edge(conn, edge.id, now, now)

        # Create deletion marker
        marker_content = reason or f"Deleted: {target.content[:100]}"
        marker = DataPointRow(
            type="memory", content=marker_content, scope=target.scope,
            salience=0.0, source_type="deletion",
            created_at=now,
        )
        marker_id = insert_data_point(conn, marker)

        # Provenance: marker supersedes deleted (uses insert_edge directly; C4 adds convenience wrapper)
        insert_edge(conn, EdgeRow(
            source=marker_id, target=id, type="supersedes",
            reason=reason, created_at=now, valid_from=now,
        ))

        conn.commit()
        return {
            "status": "deleted",
            "deleted_content": target.content,
            "marker_id": marker_id,
        }
    except Exception:
        conn.rollback()
        raise
```
