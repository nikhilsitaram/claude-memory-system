# B1: Create MCP server scaffold with mcp SDK

**Avoid:** Do not use Flask or any web framework -- the `mcp` Python SDK provides its own server class with stdio transport. Do not load the FastEmbed model synchronously at startup -- it takes 2-5 seconds and would block the MCP handshake (10-second timeout). Load it in a background thread. Do not create a new DB connection per tool call -- open one connection at startup and reuse it (SQLite WAL mode handles concurrent reads).

## Steps

### Step 2: Create scripts/memory_server.py

```python
#!/usr/bin/env python3
"""
MCP server for Claude Code Memory System.

Exposes 4 tools via the Model Context Protocol:
  - search_memories: vector + graph hybrid search
  - write_memory: atomic DB write with embedding
  - delete_memory: soft delete with provenance
  - traverse_graph: knowledge graph navigation

Runs as a persistent process managed by Claude Code.
"""
import json
import sys
import threading
from pathlib import Path

script_dir = Path(__file__).parent
if str(script_dir) not in sys.path:
    sys.path.insert(0, str(script_dir))

from mcp.server import Server
from mcp.server.stdio import stdio_server
from mcp.types import Tool, TextContent
from storage import ensure_db
from embeddings import embed_text

TOOL_NAMES = ["search_memories", "write_memory", "delete_memory", "traverse_graph"]

_db_conn = None
_model_ready = threading.Event()

def init_db():
    global _db_conn
    _db_conn = ensure_db()
    return _db_conn

def _warm_model_async():
    def _warm():
        embed_text("warmup")
        _model_ready.set()
    t = threading.Thread(target=_warm, daemon=True)
    t.start()
    return t

server = Server("memory")

@server.list_tools()
async def list_tools():
    return [
        Tool(name="search_memories", description="Search memories by semantic similarity",
             inputSchema={"type": "object", "properties": {
                 "query": {"type": "string"}, "scope": {"type": "string"},
                 "top_k": {"type": "integer", "default": 10}
             }, "required": ["query"]}),
        Tool(name="write_memory", description="Write a new memory to the knowledge base",
             inputSchema={"type": "object", "properties": {
                 "fact": {"type": "string"}, "scope": {"type": "string"},
                 "salience": {"type": "number"}, "entities": {"type": "array", "items": {"type": "string"}},
                 "supersedes": {"type": "string"}, "relation_type": {"type": "string"},
                 "relation_reason": {"type": "string"}
             }, "required": ["fact", "scope"]}),
        Tool(name="delete_memory", description="Soft-delete a memory with provenance",
             inputSchema={"type": "object", "properties": {
                 "id": {"type": "string"}, "reason": {"type": "string"}
             }, "required": ["id"]}),
        Tool(name="traverse_graph", description="Walk the knowledge graph from a data point",
             inputSchema={"type": "object", "properties": {
                 "entity": {"type": "string"}, "depth": {"type": "integer", "default": 2},
                 "relationship_type": {"type": "string"}
             }, "required": ["entity"]}),
    ]

@server.call_tool()
async def call_tool(name, arguments):
    # Stub implementations -- filled in B2-B5
    return [TextContent(type="text", text=json.dumps({"status": "not_implemented"}))]

async def main():
    init_db()
    _warm_model_async()
    async with stdio_server() as (read_stream, write_stream):
        await server.run(read_stream, write_stream, server.create_initialization_options())

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
