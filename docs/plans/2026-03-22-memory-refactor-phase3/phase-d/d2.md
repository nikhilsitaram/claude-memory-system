# D2: Create write/delete API endpoints

**Avoid:** Do not accept write/delete requests without CSRF token validation -- check the token in a custom header (e.g., `X-CSRF-Token`). Do not use GET for write operations -- POST for edit, DELETE for delete. Do not forget to re-embed after content edit (call `index_data_points` for the modified data_point).

### Step 2: Implement _edit_data_point and _delete_data_point

```python
def _edit_data_point(conn, dp_id, updates):
    dp = query_data_point_by_id(conn, dp_id)
    if not dp:
        return {"error": f"Data point {dp_id} not found"}
    if "content" in updates:
        update_data_point_content(conn, dp_id, updates["content"])
    if "salience" in updates:
        update_data_point_salience(conn, dp_id, updates["salience"])
    conn.commit()
    # Re-embed if content changed
    if "content" in updates:
        try:
            from embeddings import index_data_points, ensure_vec_table
            ensure_vec_table(conn)
            index_data_points(conn, [dp_id])
        except Exception:
            pass  # Embedding optional
    return {"status": "updated", "id": dp_id}

def _delete_data_point(conn, dp_id, reason=None):
    dp = query_data_point_by_id(conn, dp_id)
    if not dp:
        return {"error": f"Data point {dp_id} not found"}
    soft_delete_data_point(conn, dp_id)
    # Invalidate edges
    edges = query_edges_for_node(conn, dp_id)
    now = datetime.now(timezone.utc).isoformat()
    for edge in edges:
        if edge.valid_to is None:
            invalidate_edge(conn, edge.id, now, now)
    conn.commit()
    return {"status": "deleted", "id": dp_id}
```
