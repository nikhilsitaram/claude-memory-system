# A6: Archive markdown files and verify full migration

**Avoid:** Do not delete markdown files -- move them to `.archive/`. Users may want to inspect old files or roll back. Do not archive if the migration did not actually populate data_points (guard: count must be > 0). Do not archive files that are still referenced by tests -- tests use `tmp_path` fixtures and create their own files, so real filesystem archival only happens in the migration code path (not in tests).

## Steps

### Step 1: Write failing tests

```python
class TestArchiveMarkdown:
    def test_archive_moves_files(self, tmp_path):
        from storage import _archive_markdown_files
        memory_dir = tmp_path / "memory"
        daily_dir = memory_dir / "daily"
        project_dir = memory_dir / "project-memory"
        daily_dir.mkdir(parents=True)
        project_dir.mkdir(parents=True)
        (memory_dir / "global-long-term-memory.md").write_text("content")
        (daily_dir / "2026-03-20.md").write_text("daily")
        (project_dir / "myproject-long-term-memory.md").write_text("project")
        _archive_markdown_files(memory_dir)
        archive_dir = memory_dir / ".archive"
        assert archive_dir.exists()
        assert not (memory_dir / "global-long-term-memory.md").exists()
        assert not (daily_dir / "2026-03-20.md").exists()
        assert any("global-long-term-memory" in f.name for f in archive_dir.iterdir())

    def test_archive_guard_empty_db(self, tmp_path):
        """Do not archive if data_points is empty (migration failed)."""
        import sqlite3
        from storage import SCHEMA_V3_DDL, SCHEMA_DDL, _should_archive
        conn = sqlite3.connect(str(tmp_path / "test.db"))
        conn.executescript(SCHEMA_DDL)
        conn.executescript(SCHEMA_V3_DDL)
        assert _should_archive(conn) is False
        conn.close()

    def test_archive_guard_populated_db(self, tmp_path):
        """Allow archive when data_points has rows."""
        import sqlite3
        from storage import SCHEMA_V3_DDL, SCHEMA_DDL, DataPointRow, insert_data_point, _should_archive
        conn = sqlite3.connect(str(tmp_path / "test.db"))
        conn.executescript(SCHEMA_DDL)
        conn.executescript(SCHEMA_V3_DDL)
        insert_data_point(conn, DataPointRow(type="memory", content="fact", scope="global"))
        conn.commit()
        assert _should_archive(conn) is True
        conn.close()
```

### Step 2: Implement _archive_markdown_files and _should_archive

```python
def _should_archive(conn):
    count = conn.execute("SELECT COUNT(*) FROM data_points").fetchone()[0]
    return count > 0

def _archive_markdown_files(memory_dir):
    archive_dir = memory_dir / ".archive"
    archive_dir.mkdir(exist_ok=True)
    timestamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S")
    # Archive global LTM
    glm = memory_dir / "global-long-term-memory.md"
    if glm.exists():
        glm.rename(archive_dir / f"global-long-term-memory-{timestamp}.md")
    # Archive daily files
    daily_dir = memory_dir / "daily"
    if daily_dir.exists():
        for f in daily_dir.glob("*.md"):
            f.rename(archive_dir / f"daily-{f.name}")
    # Archive project LTM files
    project_dir = memory_dir / "project-memory"
    if project_dir.exists():
        for f in project_dir.glob("*-long-term-memory.md"):
            f.rename(archive_dir / f"project-{f.name}")
```
