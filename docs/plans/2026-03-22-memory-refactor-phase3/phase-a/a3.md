# A3: Implement v2-to-v3 data migration

**Avoid:** Do not use `INSERT INTO data_points SELECT * FROM chunks` -- the column sets are different (chunks has source_file, chunk_index; data_points has type, name, properties). Build explicit INSERT...SELECT with column mapping. Do not delete the old tables before verifying integrity -- the integrity check (orphan edges) must pass first. Do not run vec_data migration if sqlite-vec extension is not loadable -- guard with try/except and skip gracefully.

## Steps

### Step 1: Write failing tests for migration

```python
class TestMigrateV2ToV3:
    def _make_v2_db(self, tmp_path):
        """Create a v2 schema database with sample data."""
        import sqlite3
        from storage import SCHEMA_DDL, insert_chunk, insert_node, insert_edge, ChunkRow, NodeRow, EdgeRow
        conn = sqlite3.connect(str(tmp_path / "test.db"))
        conn.executescript(SCHEMA_DDL)
        conn.execute("PRAGMA user_version=2")
        # Insert test data
        c_id = insert_chunk(conn, ChunkRow(content="test fact", source_file="ltm.md",
                                            source_type="ltm", scope="global", entry_type="design",
                                            created_at="2026-03-20", id="chunk_abc"))
        n_id = insert_node(conn, NodeRow(name="JWT", type="library", scope="global",
                                          created_at="2026-03-20", id="node_def"))
        insert_edge(conn, EdgeRow(source=n_id, target=c_id, type="mentions",
                                   created_at="2026-03-20", id="edge_ghi"))
        conn.commit()
        return conn

    def test_chunks_become_memory_data_points(self, tmp_path):
        from storage import _migrate_v2_to_v3, query_data_point_by_id
        conn = self._make_v2_db(tmp_path)
        _migrate_v2_to_v3(conn)
        dp = query_data_point_by_id(conn, "chunk_abc")
        assert dp is not None
        assert dp.type == "memory"
        assert dp.content == "test fact"
        assert dp.scope == "global"
        conn.close()

    def test_nodes_become_entity_data_points(self, tmp_path):
        from storage import _migrate_v2_to_v3, query_data_point_by_id
        conn = self._make_v2_db(tmp_path)
        _migrate_v2_to_v3(conn)
        dp = query_data_point_by_id(conn, "node_def")
        assert dp is not None
        assert dp.type == "entity"
        assert dp.name == "JWT"
        conn.close()

    def test_edges_remain_valid_after_migration(self, tmp_path):
        from storage import _migrate_v2_to_v3
        conn = self._make_v2_db(tmp_path)
        _migrate_v2_to_v3(conn)
        orphans = conn.execute(
            "SELECT COUNT(*) FROM edges WHERE source NOT IN (SELECT id FROM data_points) "
            "OR target NOT IN (SELECT id FROM data_points)"
        ).fetchone()[0]
        assert orphans == 0
        conn.close()

    def test_edges_gain_reason_column(self, tmp_path):
        from storage import _migrate_v2_to_v3
        conn = self._make_v2_db(tmp_path)
        _migrate_v2_to_v3(conn)
        cols = {row[1] for row in conn.execute("PRAGMA table_info(edges)").fetchall()}
        assert "reason" in cols
        conn.close()

    def test_old_tables_dropped(self, tmp_path):
        from storage import _migrate_v2_to_v3
        conn = self._make_v2_db(tmp_path)
        _migrate_v2_to_v3(conn)
        tables = {row[0] for row in conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table'"
        ).fetchall()}
        assert "chunks" not in tables
        assert "nodes" not in tables
        conn.close()

    def test_migration_idempotent(self, tmp_path):
        from storage import _migrate_v2_to_v3, query_data_point_by_id
        conn = self._make_v2_db(tmp_path)
        _migrate_v2_to_v3(conn)
        _migrate_v2_to_v3(conn)  # second run should be safe
        dp = query_data_point_by_id(conn, "chunk_abc")
        assert dp is not None
        conn.close()
```

### Step 2: Implement _migrate_v2_to_v3 in storage.py

The function should:

1. Check if `data_points` table already exists (idempotent guard).
2. Create `data_points` table via `SCHEMA_V3_DDL`.
3. `INSERT INTO data_points (id, type, content, scope, ...) SELECT id, 'memory', content, scope, ... FROM chunks` -- map chunks columns to data_points. Set `type='memory'`, map `source_file` and `chunk_index` to `properties` JSON.
4. `INSERT INTO data_points (id, type, name, content, scope, ...) SELECT id, 'entity', name, description, scope, ... FROM nodes` -- map nodes columns. Set `type='entity'`, `content=description`.
5. `ALTER TABLE edges ADD COLUMN reason TEXT` (wrapped in try/except for idempotency).
6. Try to migrate vec_chunks to vec_data (guarded by sqlite-vec availability).
7. Run integrity check: orphan edge count must be 0.
8. Drop old `chunks` and `nodes` tables.
