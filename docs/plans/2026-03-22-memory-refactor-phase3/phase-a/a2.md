class TestDataPointCRUD:
    def _make_v3_db(self, tmp_path):
        """Create a v3 schema database for testing."""
        import sqlite3
        from storage import SCHEMA_V3_DDL, SCHEMA_DDL
        conn = sqlite3.connect(str(tmp_path / "test.db"))
        conn.executescript(SCHEMA_DDL)
        conn.executescript(SCHEMA_V3_DDL)
        return conn

    def test_insert_and_query_data_point(self, tmp_path):
        from storage import DataPointRow, insert_data_point, query_data_point_by_id
        conn = self._make_v3_db(tmp_path)
        dp = DataPointRow(type="memory", content="test fact", scope="global")
        dp_id = insert_data_point(conn, dp)
        conn.commit()
        result = query_data_point_by_id(conn, dp_id)
        assert result is not None
        assert result.content == "test fact"
        assert result.scope == "global"
        assert result.type == "memory"
        conn.close()

    def test_query_by_scope_with_type_filter(self, tmp_path):
        from storage import DataPointRow, insert_data_point, query_data_points_by_scope
        conn = self._make_v3_db(tmp_path)
        insert_data_point(conn, DataPointRow(type="memory", content="mem1", scope="proj"))
        insert_data_point(conn, DataPointRow(type="entity", content="ent1", scope="proj", name="JWT"))
        conn.commit()
        memories = query_data_points_by_scope(conn, "proj", dp_type="memory")
        assert len(memories) == 1
        assert memories[0].type == "memory"
        conn.close()

    def test_query_by_scope_with_min_salience(self, tmp_path):
        from storage import DataPointRow, insert_data_point, query_data_points_by_scope
        conn = self._make_v3_db(tmp_path)
        insert_data_point(conn, DataPointRow(type="memory", content="high", scope="proj", salience=0.9))
        insert_data_point(conn, DataPointRow(type="memory", content="low", scope="proj", salience=0.1))
        conn.commit()
        results = query_data_points_by_scope(conn, "proj", min_salience=0.5)
        assert len(results) == 1
        assert results[0].content == "high"
        conn.close()

    def test_soft_delete_sets_salience_zero(self, tmp_path):
        from storage import DataPointRow, insert_data_point, soft_delete_data_point, query_data_point_by_id
        conn = self._make_v3_db(tmp_path)
        dp_id = insert_data_point(conn, DataPointRow(type="memory", content="to delete", scope="global"))
        conn.commit()
        soft_delete_data_point(conn, dp_id)
        conn.commit()
        result = query_data_point_by_id(conn, dp_id)
        assert result.salience == 0.0
        conn.close()

    def test_query_nonexistent_returns_none(self, tmp_path):
