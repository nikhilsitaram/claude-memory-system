# A1: Create injection_log.py logging module with rotation

**Avoid:** Do not use `logging` module or any structured logging library -- this is a simple JSONL append. The design explicitly requires fire-and-forget with silent exception handling, not a logging framework.

**Avoid:** Do not use `fcntl` or platform-specific locking. The JSONL file is append-only and each hook invocation writes exactly one line. Concurrent appends to the same file from different processes are safe at the OS level for single-line writes under 4KB (POSIX pipe atomicity guarantee).

**Avoid:** Do not import `storage.py` or any DB modules. The injection log is a standalone JSONL file, not a SQLite table. The only import from the project should be `memory_utils.get_memory_dir` for the log file path.

#!/usr/bin/env python3
"""Injection logging for the memory system monitor.

Appends structured JSONL entries when SessionStart and UserPromptSubmit hooks
inject context into Claude Code sessions. Fire-and-forget: all public functions
catch exceptions silently so logging never breaks the hook.

Log file: ~/.claude/memory/.injection-log.jsonl
"""
import json
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

script_dir = Path(__file__).parent
if str(script_dir) not in sys.path:
    sys.path.insert(0, str(script_dir))

from memory_utils import get_memory_dir

LOG_FILENAME = ".injection-log.jsonl"
MAX_LINES = 500
KEEP_LINES = 200
CONTENT_PREVIEW_LENGTH = 80


def get_log_path() -> Path:
    """Return the path to the injection log file."""
    return get_memory_dir() / LOG_FILENAME
```

def log_session_start(
    session_id: str,
    project_scope: str,
    tiers: list[dict],
    latency_ms: float,
    health_alerts: list[str] | None = None,
) -> None:
    """Append a SessionStart injection entry to the log.

    Args:
        session_id: Claude Code session identifier.
        project_scope: Project scope name (empty string if no project).
        tiers: List of dicts, each with keys: name, count, tokens_est, ids.
        latency_ms: Time taken by _load_from_db in milliseconds.
        health_alerts: Optional list of health alert strings.
    """
    try:
        total_items = sum(t.get("count", 0) for t in tiers)
        total_tokens = sum(t.get("tokens_est", 0) for t in tiers)
        entry = {
            "ts": datetime.now(timezone.utc).astimezone().isoformat(),
            "session_id": session_id,
            "hook": "SessionStart",
            "project_scope": project_scope,
            "tiers": tiers,
            "total_items": total_items,
            "total_tokens_est": total_tokens,
            "latency_ms": round(latency_ms, 1),
            "health_alerts": health_alerts or [],
        }
        _append_entry(entry)
    except Exception:
        pass  # Fire-and-forget: never break the hook
```

def log_prompt_recall(
    session_id: str,
    prompt_preview: str,
    candidates: int,
    injected: list[dict],
    filtered: list[dict],
    latency_ms: float,
) -> None:
    """Append a UserPromptSubmit injection entry to the log.

    Args:
        session_id: Claude Code session identifier.
        prompt_preview: First 80 chars of the user's prompt.
        candidates: Total number of search results before filtering.
        injected: List of dicts with id, content_preview, scope.
        filtered: List of dicts with id, content_preview, reason.
        latency_ms: Time taken by search+filter in milliseconds.
    """
    try:
        entry = {
            "ts": datetime.now(timezone.utc).astimezone().isoformat(),
            "session_id": session_id,
            "hook": "UserPromptSubmit",
            "prompt_preview": prompt_preview[:CONTENT_PREVIEW_LENGTH],
            "candidates": candidates,
            "injected": injected,
            "filtered": filtered,
            "latency_ms": round(latency_ms, 1),
        }
        _append_entry(entry)
    except Exception:
        pass  # Fire-and-forget

### Step 4: Implement _append_entry helper

Add the private helper that both public functions use:

```python
def _append_entry(entry: dict) -> None:
    """Append a single JSON line to the injection log file.

    Creates parent directory if needed. Raises on error (caller catches).
    """
    log_path = get_log_path()
    log_path.parent.mkdir(parents=True, exist_ok=True)
    with open(log_path, "a", encoding="utf-8") as f:
        f.write(json.dumps(entry, default=str) + "\n")
```

### Step 5: Implement rotate_log

Add the rotation function:

```python
def rotate_log(max_lines: int = MAX_LINES, keep_lines: int = KEEP_LINES) -> None:
    """Rotate the injection log if it exceeds max_lines.

    Keeps the most recent keep_lines entries. Called on SessionStart.
    Silently does nothing if the file doesn't exist or is small enough.
    """
    try:
        log_path = get_log_path()
        if not log_path.exists():
            return
        lines = log_path.read_text(encoding="utf-8").splitlines()
        if len(lines) <= max_lines:
            return
        # Keep the most recent entries
        kept = lines[-keep_lines:]
        log_path.write_text("\n".join(kept) + "\n", encoding="utf-8")
    except Exception:
        pass  # Fire-and-forget
```

### Step 6: Implement read_log helper for the web API

Add a read function that the web API endpoint will use:

```python
def read_log(
    since: datetime | None = None,
    session_id: str | None = None,
    max_entries: int = 500,
) -> list[dict]:
    """Read and filter injection log entries.

    Args:
        since: Only return entries newer than this timestamp.
            Defaults to 1 hour ago if None.
        session_id: Optional session filter.
        max_entries: Maximum entries to return.

    Returns:
        List of parsed log entry dicts, chronologically ordered.
    """
    log_path = get_log_path()
    if not log_path.exists():
        return []

    if since is None:
        from datetime import timedelta
        since = datetime.now(timezone.utc).astimezone() - timedelta(hours=1)

    entries = []
    try:
        for line in log_path.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if not line:
                continue
            try:
                entry = json.loads(line)
            except json.JSONDecodeError:
                continue

            # Filter by timestamp
            entry_ts = entry.get("ts", "")
            if entry_ts:
                try:
                    ts = datetime.fromisoformat(entry_ts)
                    if ts < since:
                        continue
                except ValueError:
                    continue

            # Filter by session
            if session_id and entry.get("session_id") != session_id:
                continue

            entries.append(entry)
            if len(entries) >= max_entries:
                break
    except (IOError, OSError):
        return []

    return entries
```

class TestGetLogPath:
    def test_returns_path_with_correct_filename(self):
        with patch("injection_log.get_memory_dir", return_value=Path("/tmp/test-mem")):
            p = get_log_path()

class TestLogSessionStart:
    def test_appends_jsonl_line(self, log_dir):
        tiers = [
            {"name": "Profile", "count": 3, "tokens_est": 250, "ids": ["dp_1", "dp_2", "dp_3"]},
            {"name": "Global", "count": 2, "tokens_est": 180, "ids": ["dp_4", "dp_5"]},
        ]
        log_session_start("sess-1", "my-project", tiers, 55.3, ["stale synthesis"])

        log_path = log_dir / LOG_FILENAME
        assert log_path.exists()
        lines = log_path.read_text().splitlines()

        entry = json.loads(lines[0])

        assert entry["session_id"] == "sess-1"
        assert entry["project_scope"] == "my-project"
        assert entry["total_items"] == 5
        assert entry["total_tokens_est"] == 430
        assert entry["latency_ms"] == 55.3
        assert entry["health_alerts"] == ["stale synthesis"]
        assert len(entry["tiers"]) == 2
