class TestInjectionLogAPI:
    """Tests for GET /api/injection-log endpoint."""

        """GET /api/injection-log returns entries from the last hour."""
        from web_app import _get_injection_log
        from injection_log import log_session_start
        with patch("injection_log.get_memory_dir", return_value=tmp_path):
            log_session_start("s1", "proj", [], 10.0)
            log_session_start("s2", "proj", [], 20.0)
            result = _get_injection_log()
        assert len(result) == 2

    def test_filters_by_session(self, db, tmp_path):
        """GET /api/injection-log?session=s1 filters by session_id."""
        from web_app import _get_injection_log
        from injection_log import log_session_start
        with patch("injection_log.get_memory_dir", return_value=tmp_path):
            log_session_start("s1", "proj", [], 10.0)
            log_session_start("s2", "proj", [], 20.0)
            result = _get_injection_log(session_id="s1")
        assert len(result) == 1
        assert result[0]["session_id"] == "s1"

    def test_filters_by_since_timestamp(self, db, tmp_path):
        """GET /api/injection-log?since=<ts> filters by timestamp."""
        import json
        from datetime import datetime, timedelta, timezone
        from web_app import _get_injection_log
        from injection_log import LOG_FILENAME
        log_path = tmp_path / LOG_FILENAME
        old_ts = (datetime.now(timezone.utc) - timedelta(hours=2)).isoformat()
        new_ts = datetime.now(timezone.utc).astimezone().isoformat()
        log_path.write_text(
            json.dumps({"ts": old_ts, "session_id": "s1", "hook": "SessionStart"}) + "\n"
            + json.dumps({"ts": new_ts, "session_id": "s2", "hook": "SessionStart"}) + "\n"
        )
        with patch("injection_log.get_memory_dir", return_value=tmp_path):
            since = datetime.now(timezone.utc) - timedelta(hours=1)
            result = _get_injection_log(since=since.isoformat())
        assert len(result) == 1
        assert result[0]["session_id"] == "s2"

```python
def _get_injection_log(
    since: str | None = None,
    session_id: str | None = None,
) -> list:
    """Read injection log entries for the Monitor tab.

    Args:
        since: ISO timestamp string. Only return entries newer than this.
            Defaults to 1 hour ago.
        session_id: Optional session filter.

    Returns:
        List of parsed log entry dicts.
    """
    try:
        from injection_log import read_log
    except ImportError:
        return []

    since_dt = None
    if since:
        try:
            since_dt = datetime.fromisoformat(since)
        except ValueError:
            pass

    return read_log(since=since_dt, session_id=session_id)
```
