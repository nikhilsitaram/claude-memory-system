# B4: Add Monitor tab to web frontend

**Avoid:** Do not use any external JS libraries (no React, no Vue, no jQuery). The existing frontend uses vanilla JS with `fetch()` and inline event handlers. Follow this pattern exactly.

**Avoid:** Do not use WebSocket or Server-Sent Events. The design specifies 2-second polling via `setInterval`, paused via Page Visibility API when the tab is hidden.

**Avoid:** Do not create a separate HTML file. The entire web UI is a single `templates/web/index.html` file. Add the Monitor tab inline following the existing section pattern.

**Avoid:** Do not use `openModal()` -- the existing function is called `openDetail(dpId)`. Use that for click-to-memory navigation.

### Step 4: Add Monitor JavaScript

In the `<script>` block, before the `// Init` section (around line 800), add the Monitor JS:

```javascript
    // -----------------------------------------------------------------------
    // Monitor
    // -----------------------------------------------------------------------
    let _monitorInterval = null;
    let _monitorLastTs = null;
    let _monitorEntryIds = new Set();

    function startMonitorPolling() {
        if (_monitorInterval) return;
        _monitorLastTs = new Date(Date.now() - 3600000).toISOString(); // 1 hour ago
        _monitorEntryIds.clear();
        loadMonitor();
        _monitorInterval = setInterval(loadMonitor, 2000);
    }

    function stopMonitorPolling() {
        if (_monitorInterval) {
            clearInterval(_monitorInterval);
            _monitorInterval = null;
        }
    }

    // Pause polling when browser tab is hidden
    document.addEventListener('visibilitychange', function() {
        if (_currentSection !== 'monitor') return;
        if (document.hidden) {
            stopMonitorPolling();
        } else {
            startMonitorPolling();
        }
    });

    async function loadMonitor() {
        const session = document.getElementById('monitor-session').value;
        let url = `/api/injection-log?since=${encodeURIComponent(_monitorLastTs || '')}`;
        if (session) url += `&session=${encodeURIComponent(session)}`;

        try {
            const entries = await fetch(url).then(r => r.json());
            const timeline = document.getElementById('monitor-timeline');

            if (entries.length === 0 && timeline.children.length === 0) {
                timeline.innerHTML = '<div class="empty-state">No injection events yet. Start a Claude Code session to see activity.</div>';
                return;
            }

            // Remove empty-state message if present
            const emptyMsg = timeline.querySelector('.empty-state');
            if (emptyMsg && entries.length > 0) emptyMsg.remove();

            // Populate session dropdown with new sessions
            const sessionSelect = document.getElementById('monitor-session');
            const knownSessions = new Set(Array.from(sessionSelect.options).map(o => o.value).filter(v => v));

            for (const entry of entries) {
                const entryKey = entry.ts + entry.hook;
                if (_monitorEntryIds.has(entryKey)) continue;
                _monitorEntryIds.add(entryKey);

                // Add session to dropdown if new
                if (entry.session_id && !knownSessions.has(entry.session_id)) {
                    knownSessions.add(entry.session_id);
                    const opt = document.createElement('option');
                    opt.value = entry.session_id;
                    opt.textContent = entry.session_id.substring(0, 20);
                    sessionSelect.appendChild(opt);
                }

                // Render entry
                const el = document.createElement('div');
                el.className = 'monitor-entry';
                el.innerHTML = renderMonitorEntry(entry);
                timeline.appendChild(el);

                // Update last timestamp
                if (entry.ts > (_monitorLastTs || '')) {
                    _monitorLastTs = entry.ts;
                }
            }

            // Auto-scroll
            if (document.getElementById('monitor-autoscroll').checked && entries.length > 0) {
                timeline.scrollTop = timeline.scrollHeight;
            }
        } catch(e) {
            // Silent fail on poll errors
        }
    }

    function renderMonitorEntry(entry) {
        const time = new Date(entry.ts).toLocaleTimeString();

        if (entry.hook === 'SessionStart') {
            const hookBadge = '<span class="hook-badge hook-badge-session">SessionStart</span>';
            const latency = `<span class="monitor-latency">${entry.latency_ms}ms</span>`;
            const header = `<div class="entry-header">${hookBadge} <span>${time}</span> <span>${escHtml(entry.project_scope || 'no project')}</span> ${latency}</div>`;

            const maxTokens = Math.max(1, ...(entry.tiers || []).map(t => t.tokens_est || 0));
            const tiersHtml = (entry.tiers || []).map(tier => {
                const pct = Math.round((tier.tokens_est || 0) / maxTokens * 100);
                const idsHtml = (tier.ids || []).map(id =>
                    `<span class="monitor-id-chip" onclick="openDetail('${escHtml(id)}')">${escHtml(id.substring(0, 12))}</span>`
                ).join('');
                return `<div class="monitor-tier">
                    <span class="tier-name">${escHtml(tier.name)}</span>
                    <div class="tier-bar"><div class="tier-fill" style="width:${pct}%"></div></div>
                    <span class="tier-count">${tier.count} items / ${tier.tokens_est} tok</span>
                </div>
                ${idsHtml ? '<div class="monitor-ids">' + idsHtml + '</div>' : ''}`;
            }).join('');

            const totals = `<div style="font-size:12px;color:#a0a8c0;margin-top:6px">Total: ${entry.total_items || 0} items, ~${entry.total_tokens_est || 0} tokens</div>`;

            const alertsHtml = (entry.health_alerts || []).length > 0
                ? `<div style="color:#fb923c;font-size:12px;margin-top:4px">${entry.health_alerts.map(a => escHtml(a)).join('; ')}</div>`
                : '';

            return header + tiersHtml + totals + alertsHtml;
        }

        if (entry.hook === 'UserPromptSubmit') {
            const hookBadge = '<span class="hook-badge hook-badge-recall">PromptRecall</span>';
            const latency = `<span class="monitor-latency">${entry.latency_ms}ms</span>`;
            const header = `<div class="entry-header">${hookBadge} <span>${time}</span> <span>${entry.candidates || 0} candidates</span> ${latency}</div>`;

            const preview = `<div class="monitor-prompt-preview">"${escHtml(entry.prompt_preview || '')}"</div>`;

            const injectedHtml = (entry.injected || []).map(m =>
                `<div class="monitor-injected" onclick="openDetail('${escHtml(m.id)}')">
                    <strong>${escHtml(m.id.substring(0, 12))}</strong> (${escHtml(m.scope || '')})
                    ${escHtml(m.content_preview || '')}
                </div>`
            ).join('');

            const filteredHtml = (entry.filtered || []).map(m =>
                `<div class="monitor-filtered">
                    <strong>${escHtml(m.id.substring(0, 12))}</strong> — ${escHtml(m.reason || 'filtered')}:
                    ${escHtml(m.content_preview || '')}
                </div>`
            ).join('');

            return header + preview + injectedHtml + filteredHtml;
        }

        // Unknown hook type
        return `<div class="entry-header"><span>${time}</span> <span>${escHtml(entry.hook || 'unknown')}</span></div>`;
    }
```

### Step 5: Update showSection() to manage Monitor polling

Modify the existing `showSection` function to start/stop monitor polling:

Current (around line 437):
```javascript
    function showSection(name) {
        _currentSection = name;
        document.querySelectorAll('section').forEach(s => s.classList.remove('active'));
        document.getElementById('section-' + name).classList.add('active');
        document.querySelectorAll('#nav button').forEach(b => b.classList.remove('active'));
        event.target.classList.add('active');
        if (name === 'dashboard') loadDashboard();
        if (name === 'browse') loadBrowse(0);
        if (name === 'graph') loadGraph();
    }
```

Replace with:
```javascript
    function showSection(name) {
        // Stop monitor polling when leaving monitor tab
        if (_currentSection === 'monitor' && name !== 'monitor') {
            stopMonitorPolling();
        }

        _currentSection = name;
        document.querySelectorAll('section').forEach(s => s.classList.remove('active'));
        document.getElementById('section-' + name).classList.add('active');
        document.querySelectorAll('#nav button').forEach(b => b.classList.remove('active'));
        event.target.classList.add('active');
        if (name === 'dashboard') loadDashboard();
        if (name === 'browse') loadBrowse(0);
        if (name === 'graph') loadGraph();
        if (name === 'monitor') startMonitorPolling();
    }
```
