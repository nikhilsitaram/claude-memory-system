### Step 2: Implement get_global_working_days

Add to `scripts/memory_utils.py` after the existing `get_working_days()` function (~line 556):

```python
# Module-level cache for working-day lookups (cleared between sessions)
_working_days_cache: dict[str, list[str]] = {}


def _clear_working_days_cache() -> None:
    """Clear the working-day cache (for testing)."""
    _working_days_cache.clear()


def get_global_working_days(n: int) -> list[str]:
    """Get the last N dates with any session activity across all projects.

    Scans all project folders in ~/.claude/projects/ for .jsonl file mtimes.
    Returns dates as YYYY-MM-DD strings sorted newest-first.
    """
    cache_key = f"global:{n}"
    if cache_key in _working_days_cache:
        return _working_days_cache[cache_key]

    projects_dir = get_projects_dir()
    if not projects_dir.exists():
        _working_days_cache[cache_key] = []
        return []

    dates: set[str] = set()
    for project_folder in projects_dir.iterdir():
        if not project_folder.is_dir():
            continue
        for jsonl_file in project_folder.glob("*.jsonl"):
            mtime = datetime.fromtimestamp(
                jsonl_file.stat().st_mtime, tz=timezone.utc
            )
            dates.add(mtime.strftime("%Y-%m-%d"))

    result = sorted(dates, reverse=True)[:n]
    _working_days_cache[cache_key] = result
    return result
```

### Step 3: Implement get_project_working_days

Add below `get_global_working_days`:

```python
def get_project_working_days(project_scope: str, n: int) -> list[str]:
    """Get the last N dates with session activity for a specific project.

    Scans project folder(s) matching the scope in ~/.claude/projects/.
    Handles worktrees by checking the projects index for all folders
    mapping to the same project name.

    Args:
        project_scope: Project name (e.g., "claude-memory-system")
        n: Number of working days to return

    Returns:
        Dates as YYYY-MM-DD strings sorted newest-first.
    """
    cache_key = f"project:{project_scope}:{n}"
    if cache_key in _working_days_cache:
        return _working_days_cache[cache_key]

    projects_dir = get_projects_dir()
    if not projects_dir.exists():
        _working_days_cache[cache_key] = []
        return []

    # Find all project folders matching this scope
    index = _load_projects_index_cached()
    matching_folders: list[Path] = []
    for folder_name, info in index.items():
        if info.get("name") == project_scope:
            folder_path = projects_dir / folder_name
            if folder_path.exists():
                matching_folders.append(folder_path)

    # Fallback: if no index match, scan all folders for name suffix match
    if not matching_folders:
        for project_folder in projects_dir.iterdir():
            if project_folder.is_dir() and project_folder.name.endswith(
                f"-{project_scope}"
            ):
                matching_folders.append(project_folder)

    dates: set[str] = set()
    for folder in matching_folders:
        for jsonl_file in folder.glob("*.jsonl"):
            mtime = datetime.fromtimestamp(
                jsonl_file.stat().st_mtime, tz=timezone.utc
            )
            dates.add(mtime.strftime("%Y-%m-%d"))

    result = sorted(dates, reverse=True)[:n]
    _working_days_cache[cache_key] = result
    return result


def _load_projects_index_cached() -> dict:
    """Load projects index, using the existing cache if available."""
    global _projects_index_cache
    if _projects_index_cache is not None:
        return _projects_index_cache
    index_file = get_projects_index_file()
    return load_json_file(index_file, default={})
```

Note: `_load_projects_index_cached` reuses the existing `_projects_index_cache` global. If it's already loaded by `resolve_project_path_to_name`, we get it free. If not, load directly without populating the global (since the global cache has its own initialization logic).
