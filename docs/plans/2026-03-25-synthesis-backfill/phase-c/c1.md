```python
def run_backfill(
    days: int | None = None,
    import_from: str | None = None,
) -> int:
    """Run backfill synthesis on historical sessions.

    Args:
        days: Limit to last N calendar days (None = all sessions).
        import_from: Path to external projects directory to import first.

    Returns:
        0 on success, 1 on error.
    """
    from memory_utils import (
        get_project_working_days, load_settings, load_synthesis_state,
        update_synthesis_state,
    )
    from indexing import list_recent_sessions, get_session_date
    from storage import ensure_db

    settings = load_settings()

    # Step 1: Optional session import
    if import_from:
        from session_import import import_sessions
        result = import_sessions(import_from)
        print(f"Imported {result.copied} sessions from {result.projects} projects "
              f"({result.skipped} skipped as duplicates)")
        if result.mismatches:
            for m in result.mismatches:
                print(f"  Mismatch: {m}")

    # Step 2: Discover sessions
    max_age = days if days is not None else None
    all_sessions = list_recent_sessions(max_age_days=max_age)

    if not all_sessions:
        print("No sessions found for backfill.")
        return 0

    # Step 3: Group by project
    project_sessions = _group_sessions_by_project(all_sessions)

    # Step 4: Compute model assignments per project
    backfill_working_days = (
        settings.get("synthesis", {})
        .get("backfill", {})
        .get("recentWorkingDays", 7)
    )
    default_model = settings.get("synthesis", {}).get("model", "sonnet")

    project_batches = {}
    for project_name, sessions in project_sessions.items():
        working_days = get_project_working_days(project_name, backfill_working_days)
        working_day_set = set(working_days)

        sonnet_sessions = []
        haiku_sessions = []
        for s in sessions:
            session_date = get_session_date(s)
            if session_date in working_day_set:
                sonnet_sessions.append(s)
            else:
                haiku_sessions.append(s)

        project_batches[project_name] = {
            "sonnet": sonnet_sessions,
            "haiku": haiku_sessions,
            "total": len(sessions),
        }

    # Step 5: Print scope report
    total_sessions = sum(b["total"] for b in project_batches.values())
    total_sonnet = sum(len(b["sonnet"]) for b in project_batches.values())
    total_haiku = sum(len(b["haiku"]) for b in project_batches.values())

    print(f"\nBackfill scope: {len(project_batches)} projects, {total_sessions} sessions")
    for pname, batch in sorted(project_batches.items(), key=lambda x: -x[1]["total"]):
        print(f"  {pname:40s} {batch['total']:4d} sessions "
              f"({len(batch['sonnet'])} {default_model}, {len(batch['haiku'])} haiku)")
    print(f"\nModel split: {total_sonnet} {default_model}, {total_haiku} haiku")

    # Step 6: Ask for confirmation (stdin)
    try:
        answer = input("Proceed? [y/N] ").strip().lower()
        if answer != "y":
            print("Aborted.")
            return 0
    except EOFError:
        print("Non-interactive mode — proceeding automatically.")

    # Step 7: Process each project
    conn = ensure_db()
    state = load_synthesis_state()
    session_updates = {}

    try:
        for project_name, batch in project_batches.items():
            sonnet_count = len(batch["sonnet"])
            haiku_count = len(batch["haiku"])
            print(f"\n  {project_name} ({batch['total']} sessions)...")

            # Process sonnet batch
            if batch["sonnet"]:
                updates = _process_backfill_batch(
                    conn, batch["sonnet"], default_model, project_name, state
                )
                session_updates.update(updates)
                print(f"    {default_model}: {sonnet_count} sessions processed")

            # Process haiku batch
            if batch["haiku"]:
                updates = _process_backfill_batch(
                    conn, batch["haiku"], "haiku", project_name, state
                )
                session_updates.update(updates)
                print(f"    haiku: {haiku_count} sessions processed")

        # Step 8: Update synthesis state
        if session_updates:
            update_synthesis_state(session_updates)

        # Step 9: Run decay
        _run_decay_v3(conn)

        print(f"\nBackfill complete.")
        return 0

    except Exception as e:
        print(f"Backfill error: {e}")
        if session_updates:
            update_synthesis_state(session_updates)
        return 1
    finally:
        conn.close()
```

def _group_sessions_by_project(
    sessions: list,
) -> dict[str, list]:
    """Group sessions by resolved project name."""
    from memory_utils import resolve_project_path_to_name

    groups: dict[str, list] = {}
    for session in sessions:
        project_name = "global"  # fallback
        if session.project_path:
            resolved = resolve_project_path_to_name(session.project_path)
            if resolved:
                project_name = resolved
        elif session.project_hash:
            resolved = resolve_project_path_to_name(session.project_hash)
            if resolved:
                project_name = resolved
        groups.setdefault(project_name, []).append(session)
    return groups


def _process_backfill_batch(
    conn,
    sessions: list,
    model: str,
    project_name: str,
    state: dict,
) -> dict[str, dict]:
    """Process a batch of sessions for one project with the given model.

    Groups sessions by date, builds synthesis prompts, runs claude -p,
    applies memory ops.

    Returns dict of session_id -> {offset, lines} for state updates.
    """
    from indexing import get_session_date
    from transcript_ops import parse_jsonl_file_from_line
    from synthesis import parse_synthesis_output, apply_memory_ops_v3

    # Group by date for batching
    by_date: dict[str, list] = {}
    for s in sessions:
        day = get_session_date(s)
        by_date.setdefault(day, []).append(s)

    session_updates = {}
    sessions_state = state.get("sessions", {})

    for day, day_sessions in sorted(by_date.items()):
        # Extract transcripts for this batch
        transcripts = []
        for s in day_sessions:
            sid = s.session_id
            prev = sessions_state.get(sid)

            if prev and s.file_size == prev.get("offset", 0):
                continue  # Already synthesized

            start_line = prev.get("lines", 0) if prev else 0
            messages, total_lines = parse_jsonl_file_from_line(
                s.transcript_path, start_line=start_line
            )

            if messages:
                transcripts.append({
                    "session_id": sid,
                    "messages": messages,
                    "project_name": project_name,
                })
                session_updates[sid] = {
                    "offset": s.file_size,
                    "lines": total_lines,
                }

        if not transcripts:
            continue

        # Build and run synthesis
        prompt = _build_backfill_prompt(transcripts, project_name, day)
        output = _run_claude_synthesis(prompt, model)

        if output:
            synthesis_result = parse_synthesis_output(output)
            if synthesis_result and synthesis_result.memory_ops:
                apply_memory_ops_v3(conn, synthesis_result.memory_ops)

    return session_updates


def _build_backfill_prompt(
    transcripts: list[dict],
    project_name: str,
    day: str,
) -> str:
    """Build a synthesis prompt for a single project-date batch.

    Mirrors _build_synthesis_instructions_v3() but scoped to one project.
    """
    # Format transcript messages into the prompt
    transcript_text = []
    for t in transcripts:
        transcript_text.append(f"### Session: {t['session_id']} (project: {t['project_name']})")
        for msg in t["messages"]:
            role = msg.get("role", "unknown")
            content = msg.get("content", "")
            if isinstance(content, list):
                content = " ".join(
                    block.get("text", "") for block in content if isinstance(block, dict)
                )
            transcript_text.append(f"**{role}:** {content[:2000]}")

    instructions = _build_synthesis_instructions_v3()

    return f"""You are synthesizing session transcripts for project "{project_name}" from {day}.

{instructions}

## Transcripts

{chr(10).join(transcript_text)}
"""


def _run_claude_synthesis(prompt: str, model: str) -> str | None:
    """Run headless claude -p with the given prompt and model.

    Mirrors the subprocess pattern in _run_synthesis_v3 (lines 389-411):
    writes prompt to tempfile, pipes via stdin, 300s timeout, env CLAUDECODE="".
    """
    import os
    import tempfile

    cmd = build_claude_command(model)
    env = os.environ.copy()
    env["CLAUDECODE"] = ""  # signal non-interactive

    with tempfile.NamedTemporaryFile(mode="w", suffix=".md", delete=False) as f:
        f.write(prompt)
        prompt_path = f.name

    try:
        result = subprocess.run(
            cmd,
            input=prompt,
            capture_output=True,
            text=True,
            timeout=300,
            env=env,
        )
        if result.returncode == 0:
            return result.stdout
        return None
    except subprocess.TimeoutExpired:
        return None
    finally:
        os.unlink(prompt_path)
